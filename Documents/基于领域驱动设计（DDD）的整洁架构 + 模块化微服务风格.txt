// Fountain.SystemService.Domain/Entities/User.cs
namespace Fountain.SystemService.Domain.Entities;

public class User : BaseEntity, IAggregateRoot
{
    private readonly List<UserRole> _userRoles = new();
    
    public string UserName { get; private set; }
    public string PasswordHash { get; private set; }
    public string? Nickname { get; private set; }
    public Email? Email { get; private set; }
    public PhoneNumber? Phone { get; private set; }
    public string? Avatar { get; private set; }
    public long OrgId { get; private set; }
    public UserStatus Status { get; private set; }
    public DateTime? LastLoginTime { get; private set; }
    
    // 导航属性
    public virtual Org? Org { get; private set; }
    public IReadOnlyCollection<UserRole> UserRoles => _userRoles.AsReadOnly();
    
    private User() { } // EF Core
    
    public User(string userName, string passwordHash, long orgId)
    {
        UserName = userName;
        PasswordHash = passwordHash;
        OrgId = orgId;
        Status = UserStatus.Active;
        AddDomainEvent(new UserCreatedEvent(Id, userName));
    }
    
    public void UpdateProfile(string? nickname, Email? email, PhoneNumber? phone)
    {
        Nickname = nickname;
        Email = email;
        Phone = phone;
        AddDomainEvent(new UserProfileUpdatedEvent(Id));
    }
    
    public void ChangePassword(string newPasswordHash)
    {
        PasswordHash = newPasswordHash;
        AddDomainEvent(new UserPasswordChangedEvent(Id));
    }
    
    public void AssignRole(Role role)
    {
        if (!_userRoles.Any(ur => ur.RoleId == role.Id))
        {
            _userRoles.Add(new UserRole(Id, role.Id));
            AddDomainEvent(new UserRoleAssignedEvent(Id, role.Id));
        }
    }
    
    public void RemoveRole(Role role)
    {
        var userRole = _userRoles.FirstOrDefault(ur => ur.RoleId == role.Id);
        if (userRole != null)
        {
            _userRoles.Remove(userRole);
            AddDomainEvent(new UserRoleRemovedEvent(Id, role.Id));
        }
    }
    
    public void UpdateStatus(UserStatus status)
    {
        Status = status;
        AddDomainEvent(new UserStatusChangedEvent(Id, status));
    }
    
    public void UpdateLastLoginTime()
    {
        LastLoginTime = DateTime.UtcNow;
    }
}

// 值对象
public class Email : ValueObject
{
    public string Value { get; }
    
    public Email(string value)
    {
        if (string.IsNullOrWhiteSpace(value))
            throw new DomainException("Email cannot be empty");
        if (!IsValidEmail(value))
            throw new DomainException("Invalid email format");
        Value = value;
    }
    
    private static bool IsValidEmail(string email)
    {
        try
        {
            var addr = new System.Net.Mail.MailAddress(email);
            return addr.Address == email;
        }
        catch
        {
            return false;
        }
    }
    
    protected override IEnumerable<object> GetEqualityComponents()
    {
        yield return Value;
    }
}

// Fountain.SystemService.Application/Features/Users/Commands/CreateUser/CreateUserCommand.cs
namespace Fountain.SystemService.Application.Features.Users.Commands.CreateUser;

public record CreateUserCommand : IRequest<Result<long>>
{
    public string UserName { get; init; }
    public string Password { get; init; }
    public string? Nickname { get; init; }
    public string? Email { get; init; }
    public string? Phone { get; init; }
    public long OrgId { get; init; }
    public List<long> RoleIds { get; init; } = new();
}

// 命令处理器
public class CreateUserCommandHandler : IRequestHandler<CreateUserCommand, Result<long>>
{
    private readonly IUserRepository _userRepository;
    private readonly IRoleRepository _roleRepository;
    private readonly ICurrentUserService _currentUserService;
    private readonly IPasswordHasher _passwordHasher;
    private readonly IEventBus _eventBus;
    private readonly IMapper _mapper;
    
    public CreateUserCommandHandler(
        IUserRepository userRepository,
        IRoleRepository roleRepository,
        ICurrentUserService currentUserService,
        IPasswordHasher passwordHasher,
        IEventBus eventBus,
        IMapper mapper)
    {
        _userRepository = userRepository;
        _roleRepository = roleRepository;
        _currentUserService = currentUserService;
        _passwordHasher = passwordHasher;
        _eventBus = eventBus;
        _mapper = mapper;
    }
    
    public async Task<Result<long>> Handle(CreateUserCommand request, CancellationToken cancellationToken)
    {
        // 检查用户名唯一性
        var existingUser = await _userRepository.GetByUserNameAsync(request.UserName, cancellationToken);
        if (existingUser != null)
            return Result<long>.Error($"用户名 {request.UserName} 已存在");
        
        // 创建用户实体
        var passwordHash = _passwordHasher.Hash(request.Password);
        var user = new User(
            request.UserName,
            passwordHash,
            request.OrgId);
        
        // 设置基本信息
        user.UpdateProfile(request.Nickname, 
            request.Email != null ? new Email(request.Email) : null,
            request.Phone != null ? new PhoneNumber(request.Phone) : null);
        
        // 分配角色
        if (request.RoleIds.Any())
        {
            var roles = await _roleRepository.GetByIdsAsync(request.RoleIds, cancellationToken);
            foreach (var role in roles)
            {
                user.AssignRole(role);
            }
        }
        
        // 设置审计信息
        user.SetCreatedBy(_currentUserService.UserId, _currentUserService.UserName);
        
        // 保存
        await _userRepository.AddAsync(user, cancellationToken);
        await _userRepository.SaveChangesAsync(cancellationToken);
        
        // 发布领域事件
        foreach (var domainEvent in user.DomainEvents)
        {
            await _eventBus.Publish(domainEvent, cancellationToken);
        }
        user.ClearDomainEvents();
        
        return Result<long>.Success(user.Id);
    }
}

// 验证器
public class CreateUserCommandValidator : AbstractValidator<CreateUserCommand>
{
    public CreateUserCommandValidator()
    {
        RuleFor(x => x.UserName)
            .NotEmpty().WithMessage("用户名不能为空")
            .MinimumLength(3).WithMessage("用户名至少3个字符")
            .MaximumLength(50).WithMessage("用户名最多50个字符");
            
        RuleFor(x => x.Password)
            .NotEmpty().WithMessage("密码不能为空")
            .MinimumLength(6).WithMessage("密码至少6个字符")
            .Matches(@"^(?=.*[A-Za-z])(?=.*\d)[A-Za-z\d]{6,}$")
            .WithMessage("密码必须包含字母和数字");
            
        RuleFor(x => x.Email)
            .EmailAddress().When(x => !string.IsNullOrEmpty(x.Email))
            .WithMessage("邮箱格式不正确");
            
        RuleFor(x => x.Phone)
            .Matches(@"^1[3-9]\d{9}$").When(x => !string.IsNullOrEmpty(x.Phone))
            .WithMessage("手机号格式不正确");
    }
}

namespace Fountain.SystemService.Domain.Repositories;

public interface IUserRepository : IRepository<User>

    Task<PaginatedList<User>> GetPagedAsync(UserSpecification spec, int pageIndex, int pageSize, CancellationToken cancellationToken);
    Task<bool> IsUserNameUniqueAsync(string userName, long? excludeId = null, CancellationToken cancellationToken = default);
}

// Fountain.SystemService.Infrastructure/Persistence/Repositories/UserRepository.cs
namespace Fountain.SystemService.Infrastructure.Persistence.Repositories;

public class UserRepository : RepositoryBase<User>, IUserRepository
{
    public UserRepository(ApplicationDbContext dbContext) : base(dbContext)
    {
    }
    
    public async Task<User?> GetByUserNameAsync(string userName, CancellationToken cancellationToken)
    {
        return await _dbContext.Users
            .Include(x => x.UserRoles)
            .ThenInclude(ur => ur.Role)
            .FirstOrDefaultAsync(x => x.UserName == userName && !x.IsDeleted, cancellationToken);
    }
    
    public async Task<User?> GetByIdWithRolesAsync(long id, CancellationToken cancellationToken)
    {
        return await _dbContext.Users
            .Include(x => x.UserRoles)
            .ThenInclude(ur => ur.Role)
            .FirstOrDefaultAsync(x => x.Id == id && !x.IsDeleted, cancellationToken);
    }
    
    public async Task<PaginatedList<User>> GetPagedAsync(UserSpecification spec, int pageIndex, int pageSize, CancellationToken cancellationToken)
    {
        var query = _dbContext.Users
            .Include(x => x.Org)
            .Include(x => x.UserRoles)
            .ThenInclude(ur => ur.Role)
            .AsNoTracking();
            
        query = spec.Apply(query);
        query = query.OrderByDescending(x => x.CreatedTime);
        
        var totalCount = await query.CountAsync(cancellationToken);
        var items = await query.Skip((pageIndex - 1) * pageSize)
            .Take(pageSize)
            .ToListAsync(cancellationToken);
            
        return new PaginatedList<User>(items, totalCount, pageIndex, pageSize);
    }
}

// Fountain.SystemService.Api/Controllers/v1/UsersController.cs
namespace Fountain.SystemService.Api.Controllers.v1;

[ApiController]
[ApiVersion("1.0")]
[Route("api/v{version:apiVersion}/[controller]")]
[Authorize]
public class UsersController : BaseApiController
{
    /// <summary>
    /// 获取用户列表
    /// </summary>
    [HttpGet]
    [Permission("user:list")]
    [ProducesResponseType(typeof(Result<PaginatedList<UserDto>>), StatusCodes.Status200OK)]
    public async Task<ActionResult<Result<PaginatedList<UserDto>>>> GetList([FromQuery] GetUserListQuery query)
    {
        return Ok(await Mediator.Send(query));
    }
    
    /// <summary>
    /// 获取用户详情
    /// </summary>
    [HttpGet("{id}")]
    [Permission("user:view")]
    [ProducesResponseType(typeof(Result<UserDto>), StatusCodes.Status200OK)]
    [ProducesResponseType(typeof(Result), StatusCodes.Status404NotFound)]
    public async Task<ActionResult<Result<UserDto>>> GetById(long id)
    {
        return Ok(await Mediator.Send(new GetUserByIdQuery { Id = id }));
    }
    
    /// <summary>
    /// 创建用户
    /// </summary>
    [HttpPost]
    [Permission("user:add")]
    [ProducesResponseType(typeof(Result<long>), StatusCodes.Status201Created)]
    [ProducesResponseType(typeof(Result), StatusCodes.Status400BadRequest)]
    public async Task<ActionResult<Result<long>>> Create([FromBody] CreateUserCommand command)
    {
        var result = await Mediator.Send(command);
        return CreatedAtAction(nameof(GetById), new { id = result.Data }, result);
    }
    
    /// <summary>
    /// 更新用户
    /// </summary>
    [HttpPut("{id}")]
    [Permission("user:edit")]
    [ProducesResponseType(typeof(Result), StatusCodes.Status200OK)]
    [ProducesResponseType(typeof(Result), StatusCodes.Status400BadRequest)]
    public async Task<ActionResult<Result>> Update(long id, [FromBody] UpdateUserCommand command)
    {
        if (id != command.Id)
            return BadRequest(Result.Error("路由ID与请求体ID不匹配"));
            
        return Ok(await Mediator.Send(command));
    }
    
    /// <summary>
    /// 删除用户
    /// </summary>
    [HttpDelete("{id}")]
    [Permission("user:delete")]
    [ProducesResponseType(typeof(Result), StatusCodes.Status200OK)]
    [ProducesResponseType(typeof(Result), StatusCodes.Status404NotFound)]
    public async Task<ActionResult<Result>> Delete(long id)
    {
        return Ok(await Mediator.Send(new DeleteUserCommand { Id = id }));
    }
    
    /// <summary>
    /// 分配角色
    /// </summary>
    [HttpPost("{id}/roles")]
    [Permission("user:assignRole")]
    public async Task<ActionResult<Result>> AssignRoles(long id, [FromBody] AssignUserRoleCommand command)
    {
        if (id != command.UserId)
            return BadRequest(Result.Error("路由ID与请求体用户ID不匹配"));
            
        return Ok(await Mediator.Send(command));
    }

namespace Fountain.EventBus.Abstractions;

public interface IEventBus
{
    Task Publish<T>(T @event, CancellationToken cancellationToken = default) where T : class;
    Task Subscribe<T, TH>()
        where T : class
        where TH : IIntegrationEventHandler<T>;
}

// 领域事件
public record UserCreatedEvent : DomainEvent
{
    public long UserId { get; }
    public string UserName { get; }
    
    public UserCreatedEvent(long userId, string userName)
    {
        UserId = userId;
        UserName = userName;
    }
}

// 集成事件
public record UserCreatedIntegrationEvent : IntegrationEvent
{
    public long UserId { get; }
    public string UserName { get; }
    public string? Email { get; }
    
    public UserCreatedIntegrationEvent(long userId, string userName, string? email)
    {
        UserId = userId;
        UserName = userName;
        Email = email;
    }
}

// 事件处理器
public class UserCreatedDomainEventHandler : INotificationHandler<UserCreatedEvent>
{
    private readonly IEventBus _eventBus;
    private readonly ILogger<UserCreatedDomainEventHandler> _logger;
    
    public UserCreatedDomainEventHandler(IEventBus eventBus, ILogger<UserCreatedDomainEventHandler> logger)
    {
        _eventBus = eventBus;
        _logger = logger;
    }
    
    public async Task Handle(UserCreatedEvent notification, CancellationToken cancellationToken)
    {
        _logger.LogInformation("User created: {UserId} - {UserName}", notification.UserId, notification.UserName);
        
        // 发布集成事件到消息队列
        var integrationEvent = new UserCreatedIntegrationEvent(
            notification.UserId, 
            notification.UserName, 
            null);
            
        await _eventBus.Publish(integrationEvent, cancellationToken);
    }
}

// Fountain.Shared/Domain/BaseEntity.cs
namespace Fountain.Shared.Domain;

public abstract class BaseEntity
{
    private readonly List<DomainEvent> _domainEvents = new();
    
    public long Id { get; protected set; }
    public bool IsDeleted { get; protected set; }
    
    // 审计字段
    public long CreatedBy { get; protected set; }
    public string? CreatedByName { get; protected set; }
    public DateTime CreatedTime { get; protected set; }
    public long UpdatedBy { get; protected set; }
    public string? UpdatedByName { get; protected set; }
    public DateTime UpdatedTime { get; protected set; }
    public long? DeletedBy { get; protected set; }
    public string? DeletedByName { get; protected set; }
    public DateTime? DeletedTime { get; protected set; }
    
    public IReadOnlyCollection<DomainEvent> DomainEvents => _domainEvents.AsReadOnly();
    
    protected void AddDomainEvent(DomainEvent domainEvent)
    {
        _domainEvents.Add(domainEvent);
    }
    
    public void ClearDomainEvents()
    {
        _domainEvents.Clear();
    }
    
    public void SetCreatedBy(long userId, string? userName)
    {
        CreatedBy = userId;
        CreatedByName = userName;
        CreatedTime = DateTime.UtcNow;
        UpdatedBy = userId;
        UpdatedByName = userName;
        UpdatedTime = DateTime.UtcNow;
    }
    
    public void SetUpdatedBy(long userId, string? userName)
    {
        UpdatedBy = userId;
        UpdatedByName = userName;
        UpdatedTime = DateTime.UtcNow;
    }
    
    public void SoftDelete(long userId, string? userName)
    {
        IsDeleted = true;
        DeletedBy = userId;
        DeletedByName = userName;
        DeletedTime = DateTime.UtcNow;
        SetUpdatedBy(userId, userName);
    }
}
