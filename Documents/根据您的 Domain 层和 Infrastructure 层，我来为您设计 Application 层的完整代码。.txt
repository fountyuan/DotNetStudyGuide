```csharp
/// <summary>
/// API响应基类
/// </summary>
public class ApiResponse<T>
{
    public bool Success { get; set; }
    public string? Message { get; set; }
    public string? Code { get; set; }
    public T? Data { get; set; }
    public DateTime Timestamp { get; set; } = DateTime.UtcNow;

    public static ApiResponse<T> Ok(T? data = default, string? message = null)
    {
        return new ApiResponse<T>
        {
            Success = true,
            Data = data,
            Message = message ?? "操作成功"
        };
    }

    public static ApiResponse<T> Fail(string message, string? code = null)
    {
        return new ApiResponse<T>
        {
            Success = false,
            Message = message,
            Code = code
        };
    }
}
```

```csharp
namespace Fountain.Service.Identity.Application.DTOs.Common;

/// <summary>
/// 分页结果
/// </summary>
public class PaginatedResult<T>
{
    public List<T> Items { get; set; } = new();
    public int PageIndex { get; set; }
    public int PageSize { get; set; }
    public long TotalCount { get; set; }
    public int TotalPages => (int)Math.Ceiling(TotalCount / (double)PageSize);
    public bool HasPreviousPage => PageIndex > 1;
    public bool HasNextPage => PageIndex < TotalPages;

    public PaginatedResult() { }

    public PaginatedResult(List<T> items, long totalCount, int pageIndex, int pageSize)
    {
        Items = items;
        TotalCount = totalCount;
        PageIndex = pageIndex;
        PageSize = pageSize;
    }
}
```

## 4. Handlers

### Users/CreateUserCommandHandler.cs

```csharp
using MediatR;
using AutoMapper;
using Fountain.Service.Identity.Application.Commands.Users;
using Fountain.Service.Identity.Application.DTOs.Users;
using Fountain.Service.Identity.Domain.Entities;
using Fountain.Service.Identity.Domain.Repositories;
using Fountain.Service.Identity.Domain.ValueObjects;

namespace Fountain.Service.Identity.Application.Handlers.Users;

/// <summary>
/// 创建用户命令处理器
/// </summary>
public class CreateUserCommandHandler : IRequestHandler<CreateUserCommand, UserDto>
{
    private readonly IUserRepository _userRepository;
    private readonly IRoleRepository _roleRepository;
    private readonly IOrganizationRepository _organizationRepository;
    private readonly IMapper _mapper;
    private readonly ICurrentUser _currentUser;

    public CreateUserCommandHandler(
        IUserRepository userRepository,
        IRoleRepository roleRepository,
        IOrganizationRepository organizationRepository,
        IMapper mapper,
        ICurrentUser currentUser)
    {
        _userRepository = userRepository;
        _roleRepository = roleRepository;
        _organizationRepository = organizationRepository;
        _mapper = mapper;
        _currentUser = currentUser;
    }

    public async Task<UserDto> Handle(CreateUserCommand request, CancellationToken cancellationToken)
    {
        var dto = request.Data;

        // 验证用户名唯一性
        var exists = await _userRepository.ExistsByUsernameAsync(dto.Username, cancellationToken: cancellationToken);
        if (exists)
            throw new InvalidOperationException($"用户名 {dto.Username} 已存在");

        // 验证邮箱唯一性
        if (!string.IsNullOrEmpty(dto.Email))
        {
            exists = await _userRepository.ExistsByEmailAsync(dto.Email, cancellationToken: cancellationToken);
            if (exists)
                throw new InvalidOperationException($"邮箱 {dto.Email} 已被使用");
        }

        // 验证手机号唯一性
        if (!string.IsNullOrEmpty(dto.Phone))
        {
            exists = await _userRepository.ExistsByPhoneAsync(dto.Phone, cancellationToken: cancellationToken);
            if (exists)
                throw new InvalidOperationException($"手机号 {dto.Phone} 已被使用");
        }

        // 验证组织存在
        var org = await _organizationRepository.GetByIdAsync(dto.OrgId, cancellationToken);
        if (org == null)
            throw new InvalidOperationException($"组织 {dto.OrgId} 不存在");

        // 创建用户
        var email = !string.IsNullOrEmpty(dto.Email) ? new Email(dto.Email) : null;
        var phone = !string.IsNullOrEmpty(dto.Phone) ? new PhoneNumber(dto.Phone) : null;
        
        var user = new User(
            dto.Username,
            dto.OrgId,
            dto.Nickname,
            email,
            phone,
            createdBy: _currentUser.Id,
            createdByName: _currentUser.Name ?? dto.Username);

        // 设置密码
        user.SetPassword(dto.Password);

        // 分配角色
        foreach (var roleId in dto.RoleIds)
        {
            var role = await _roleRepository.GetByIdAsync(roleId, cancellationToken);
            if (role != null)
            {
                user.AssignRole(roleId);
            }
        }

        // 保存
        var created = await _userRepository.AddAsync(user, cancellationToken);
        
        return _mapper.Map<UserDto>(created);
    }
}
```

### Users/GetUserListQueryHandler.cs

```csharp
using MediatR;
using AutoMapper;
using Fountain.Framework.Core.Specifications.Query;
using Fountain.Service.Identity.Application.DTOs.Common;
using Fountain.Service.Identity.Application.DTOs.Users;
using Fountain.Service.Identity.Application.Queries.Users;
using Fountain.Service.Identity.Domain.Entities;
using Fountain.Service.Identity.Domain.Repositories;
using Fountain.Service.Identity.Domain.Services;

namespace Fountain.Service.Identity.Application.Handlers.Users;

/// <summary>
/// 获取用户列表查询处理器
/// </summary>
public class GetUserListQueryHandler : IRequestHandler<GetUserListQuery, PaginatedResult<UserListDto>>
{
    private readonly IUserRepository _userRepository;
    private readonly IDataScopeService _dataScopeService;
    private readonly IMapper _mapper;
    private readonly ICurrentUser _currentUser;

    public GetUserListQueryHandler(
        IUserRepository userRepository,
        IDataScopeService dataScopeService,
        IMapper mapper,
        ICurrentUser currentUser)
    {
        _userRepository = userRepository;
        _dataScopeService = dataScopeService;
        _mapper = mapper;
        _currentUser = currentUser;
    }

    public async Task<PaginatedResult<UserListDto>> Handle(GetUserListQuery request, CancellationToken cancellationToken)
    {
        // 获取当前用户的数据权限范围
        var authorizedOrgIds = await _dataScopeService.GetAuthorizedOrgIdsAsync(_currentUser.Id, cancellationToken);
        
        // 构建规约
        var spec = new PaginationSpecification<User>();
        
        // 添加组织权限过滤
        if (authorizedOrgIds.Any())
        {
            spec.Conditions.Add(u => authorizedOrgIds.Contains(u.OrgId));
        }
        
        // 添加关键词过滤
        if (!string.IsNullOrEmpty(request.Keyword))
        {
            spec.Conditions.Add(u => 
                u.Username.Contains(request.Keyword) || 
                (u.Nickname != null && u.Nickname.Contains(request.Keyword)) ||
                (u.Email != null && u.Email.Value.Contains(request.Keyword)) ||
                (u.Phone != null && u.Phone.Value.Contains(request.Keyword)));
        }
        
        // 添加组织过滤
        if (request.OrgId.HasValue)
        {
            spec.Conditions.Add(u => u.OrgId == request.OrgId.Value);
        }
        
        // 添加状态过滤
        if (!string.IsNullOrEmpty(request.Status))
        {
            // 状态过滤逻辑
        }
        
        // 分页查询
        var (users, total) = await _userRepository.GetPagedAsync(spec, request.PageIndex, request.PageSize, cancellationToken);
        
        var dtos = _mapper.Map<List<UserListDto>>(users);
        
        return new PaginatedResult<UserListDto>(dtos, total, request.PageIndex, request.PageSize);
    }
}
```

### Auth/LoginCommandHandler.cs

```csharp
using MediatR;
using AutoMapper;
using Fountain.Service.Identity.Application.Commands.Auth;
using Fountain.Service.Identity.Application.DTOs.Auth;
using Fountain.Service.Identity.Application.DTOs.Menus;
using Fountain.Service.Identity.Domain.Services;

namespace Fountain.Service.Identity.Application.Handlers.Auth;

/// <summary>
/// 登录命令处理器
/// </summary>
public class LoginCommandHandler : IRequestHandler<LoginCommand, LoginResponseDto>
{
    private readonly IAuthService _authService;
    private readonly IPermissionService _permissionService;
    private readonly IMapper _mapper;

    public LoginCommandHandler(
        IAuthService authService,
        IPermissionService permissionService,
        IMapper mapper)
    {
        _authService = authService;
        _permissionService = permissionService;
        _mapper = mapper;
    }

    public async Task<LoginResponseDto> Handle(LoginCommand request, CancellationToken cancellationToken)
    {
        var (user, token, refreshToken) = await _authService.LoginByPasswordAsync(
            request.Data.Account,
            request.Data.Password,
            request.IpAddress,
            request.UserAgent,
            cancellationToken);

        // 获取用户权限
        var permissions = await _permissionService.GetUserPermissionsAsync(user.Id, cancellationToken);
        
        // 获取用户菜单
        var menus = await _permissionService.GetUserMenusAsync(user.Id, cancellationToken);
        var menuDtos = _mapper.Map<List<MenuTreeDto>>(menus);

        return new LoginResponseDto
        {
            UserId = user.Id,
            Username = user.Username,
            Nickname = user.Nickname,
            Avatar = user.Avatar,
            Token = token,
            RefreshToken = refreshToken.Token,
            ExpiresAt = refreshToken.ExpiresAt,
            Permissions = permissions.Select(p => p.ToString()).ToList(),
            Menus = menuDtos
        };
    }
}
```

## 5. Mapping Profile

```csharp
using AutoMapper;
using Fountain.Service.Identity.Application.DTOs.Menus;
using Fountain.Service.Identity.Application.DTOs.Organizations;
using Fountain.Service.Identity.Application.DTOs.Roles;
using Fountain.Service.Identity.Application.DTOs.Users;
using Fountain.Service.Identity.Domain.Entities;

namespace Fountain.Service.Identity.Application.Common.Mappings;

/// <summary>
/// AutoMapper配置
/// </summary>
public class MappingProfile : Profile
{
    public MappingProfile()
    {
        // User mappings
        CreateMap<User, UserDto>()
            .ForMember(dest => dest.Email, opt => opt.MapFrom(src => src.Email != null ? src.Email.Value : null))
            .ForMember(dest => dest.Phone, opt => opt.MapFrom(src => src.Phone != null ? src.Phone.Value : null))
            .ForMember(dest => dest.Status, opt => opt.MapFrom(src => src.Status.ToString()))
            .ForMember(dest => dest.UserType, opt => opt.MapFrom(src => src.UserType.ToString()))
            .ForMember(dest => dest.RoleIds, opt => opt.MapFrom(src => src.UserRoles.Select(ur => ur.RoleId)))
            .ForMember(dest => dest.RoleNames, opt => opt.Ignore());

        CreateMap<User, UserListDto>()
            .ForMember(dest => dest.Email, opt => opt.MapFrom(src => src.Email != null ? src.Email.Value : null))
            .ForMember(dest => dest.Phone, opt => opt.MapFrom(src => src.Phone != null ? src.Phone.Value : null))
            .ForMember(dest => dest.Status, opt => opt.MapFrom(src => src.Status.ToString()))
            .ForMember(dest => dest.OrgName, opt => opt.Ignore());

        // Role mappings
        CreateMap<Role, RoleDto>()
            .ForMember(dest => dest.DataScopeType, opt => opt.MapFrom(src => src.DataScope.Scope.ToString()))
            .ForMember(dest => dest.CustomOrgIds, opt => opt.MapFrom(src => src.DataScope.CustomOrgIds))
            .ForMember(dest => dest.MenuIds, opt => opt.MapFrom(src => src.RoleMenus.Select(rm => rm.MenuId)));

        // Menu mappings
        CreateMap<Menu, MenuDto>();
        CreateMap<Menu, MenuTreeDto>()
            .ForMember(dest => dest.MenuType, opt => opt.MapFrom(src => src.MenuType.ToString()))
            .ForMember(dest => dest.Children, opt => opt.MapFrom(src => src.Children));

        // Organization mappings
        CreateMap<Organization, OrganizationDto>()
            .ForMember(dest => dest.OrgType, opt => opt.MapFrom(src => src.OrgType.ToString()));
            
        CreateMap<Organization, OrganizationTreeDto>()
            .ForMember(dest => dest.OrgType, opt => opt.MapFrom(src => src.OrgType.ToString()))
            .ForMember(dest => dest.Children, opt => opt.MapFrom(src => src.Children));
    }
```

## 7. CurrentUserService Implementation

```csharp
using System.Security.Claims;
using Fountain.Service.Identity.Application.Common.Interfaces;
using Fountain.Service.Identity.Domain.Services;
using Microsoft.AspNetCore.Http;

namespace Fountain.Service.Identity.Application.Services;

/// <summary>
/// 当前用户服务实现
/// </summary>
public class CurrentUserService : ICurrentUser
{
    private readonly IHttpContextAccessor _httpContextAccessor;
    private readonly IPermissionService _permissionService;
    private ClaimsPrincipal? _user;

    public CurrentUserService(IHttpContextAccessor httpContextAccessor, IPermissionService permissionService)
    {
        _httpContextAccessor = httpContextAccessor;
        _permissionService = permissionService;
        _user = _httpContextAccessor.HttpContext?.User;
    }

    public long Id => GetClaimValue<long>(ClaimTypes.NameIdentifier);
    public string? Name => GetClaimValue<string>(ClaimTypes.Name);
    public string? UserName => GetClaimValue<string>("preferred_username");
    public string? Email => GetClaimValue<string>(ClaimTypes.Email);
    public string? Phone => GetClaimValue<string>(ClaimTypes.MobilePhone);
    public long OrgId => GetClaimValue<long>("org_id");
    public bool IsAuthenticated => _user?.Identity?.IsAuthenticated ?? false;

    public bool HasPermission(string permission)
    {
        var permissions = GetPermissions();
        return permissions.Contains(permission);
    }

    public async Task<bool> HasPermissionAsync(string permission)
    {
        return await _permissionService.CheckPermissionAsync(Id, permission);
    }

    public List<string> GetPermissions()
    {
        var permissionClaim = _user?.FindFirst("permissions");
        if (permissionClaim == null)
            return new List<string>();
        
        return permissionClaim.Value.Split(',').ToList();
    }

    private T GetClaimValue<T>(string claimType)
    {
        var claim = _user?.FindFirst(claimType);
        if (claim == null)
            return default!;

        return (T)Convert.ChangeType(claim.Value, typeof(T));
    }
}
```

## 9. Dependency Injection Extension

```csharp
using FluentValidation;
using MediatR;
using Microsoft.Extensions.DependencyInjection;
using System.Reflection;
using Fountain.Service.Identity.Application.Common.Interfaces;
using Fountain.Service.Identity.Application.Services;

namespace Fountain.Service.Identity.Application;

/// <summary>
/// 应用层依赖注入扩展
/// </summary>
public static class DependencyInjection
{
    public static IServiceCollection AddApplication(this IServiceCollection services)
    {
        // 注册 MediatR
        services.AddMediatR(cfg => {
            cfg.RegisterServicesFromAssembly(Assembly.GetExecutingAssembly());
        });

        // 注册 AutoMapper
        services.AddAutoMapper(Assembly.GetExecutingAssembly());

        // 注册 FluentValidation
        services.AddValidatorsFromAssembly(Assembly.GetExecutingAssembly());

        // 注册当前用户服务
        services.AddScoped<ICurrentUser, CurrentUserService>();
        services.AddHttpContextAccessor();

        return services;
    }
}
```

```csharp
    }

    public async Task<ActionResult<ApiResponse<UserDto>>> Update(long id, [FromBody] UpdateUserCommand command)

        return Ok(ApiResponse<UserDto>.Ok(result, "更新成功"));
    }

    }

    }
```
